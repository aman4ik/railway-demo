      <input type="text" id="dataInput" placeholder="Введите новую запись...">

      <button id="addBtn">Добавить</button>

    </div>

    <ul id="list"></ul>

  </div>


  <script type="module">

    // Подтягиваем ссылку на бэкенд из настроек Railway

    const API_URL = import.meta.env.VITE_API_URL || '';

    

    const input = document.getElementById('dataInput');

    const btn = document.getElementById('addBtn');

    const list = document.getElementById('list');


    // Функция получения данных из базы

    async function fetchRecords() {

      try {

        const res = await fetch(`${API_URL}/api/data`);

        const data = await res.json();

        list.innerHTML = '';

        data.forEach(item => {

          const li = document.createElement('li');

          // В зависимости от того, как написан твой Python-бэкенд, тут может быть item.data, item.text или item.content

          li.textContent = item.data || item.text || item.content || JSON.stringify(item);

          list.appendChild(li);

        });

      } catch (e) {

        console.error('Ошибка загрузки списка:', e);

      }

    }


    // Функция добавления новой записи

    async function addRecord() {

      const text = input.value.trim();

      if (!text) return;

      

      try {

        await fetch(`${API_URL}/api/data`, {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          // Убедись, что ключ ("data") совпадает с тем, что ждет твой Python-бэкенд

          body: JSON.stringify({ data: text }) 

        });

        input.value = '';

        fetchRecords();

      } catch (e) {

        console.error('Ошибка добавления:', e);

        alert('Не удалось добавить запись. Нажми F12 (Inspect) и проверь консоль!');

      }

    }


    btn.addEventListener('click', addRecord);

    input.addEventListener('keypress', (e) => {

      if (e.key === 'Enter') addRecord();

    });


    // Загружаем список при открытии страницы

    fetchRecords();

  </script>
